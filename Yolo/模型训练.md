# 训练代码

```python
from ultralytics import YOLO

if __name__ == '__main__':
    # 1. 加载预训练模型
    model = YOLO('yolov8n.pt')

    # 2. 开始训练
    model.train(
        data='yolo-bvn.yaml',
        workers=1,       # workers 是让 CPU 多派几个"搬运工"给 GPU 喂数据，1 是最保守设置，正常用 4~8 更高效
        epochs=300,      # 上限设大一点
        patience=50,     # 连续 50 轮 mAP 无提升就自动停止
        batch=16,		 # 一次性喂给模型16张图片
        name='train',	 # 结果存放文件名
        exist_ok=True	 # 结果允许覆盖
    )
```



# 使用模型

yolo detect predict model=runs/detect/train/weight/best.pt source=() show=True
